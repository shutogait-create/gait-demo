function doPost(e) {
  var body;
  try { body = JSON.parse(e.postData.contents); } catch (err) { return ContentService.createTextOutput('bad request'); }

  // LINEからのWebhook（events配列がある）はこちらで処理
  if (body && body.events) { return handleLineWebhook_(body); }

  return handleSupabaseHook_(body, e);
}

function handleSupabaseHook_(body, e) {
  try {
    if (body.secret !== SHARED_SECRET) {
      return ContentService.createTextOutput('forbidden');
    }

    var title    = body.title    || 'gait ご予約';
    var start    = new Date(body.start);
    var minutes  = Number(body.duration_min) || 60;
    var end      = new Date(start.getTime() + minutes * 60000);
    var customer = body.customer || '';
    var menu     = body.menu     || '';
    var staff    = body.staff    || '';
    var channel  = body.channel  || 'app';
    var action   = body.action   || 'created';
    var bookingId= body.booking_id || '';

    var cal = CalendarApp.getCalendarById(CALENDAR_ID);
    var desc = [
      '顧客：' + customer,
      'メニュー：' + menu,
      '担当：' + staff,
      '経路：' + channel,
      '',
      APP_MARKER + 'booking:' + bookingId,   // ← この行があるとカレンダー→アプリ取り込みの対象外になります
      'オーナー管理：https://app.gait-pilates.com/owner.html'
    ].join('\n');

    if (action === 'cancelled') {
      var evs = cal.getEvents(new Date(start.getTime() - 60000), new Date(end.getTime() + 60000));
      for (var i = 0; i < evs.length; i++) {
        if (evs[i].getTitle().indexOf(customer) >= 0) { evs[i].deleteEvent(); break; }
      }
      MailApp.sendEmail(NOTIFY_EMAIL,
        '【gait】予約キャンセル ' + fmt(start),
        '予約がキャンセルされました。\n\n' + fmt(start) + '\n' + desc);
      return ContentService.createTextOutput('cancelled');
    }

    cal.createEvent(title, start, end, { description: desc, location: STUDIO });

    // お客様のLINEへ予約確定メッセージ（連携済みの方だけ）
    if (body.line_user_id) {
      var okLine = linePush_(body.line_user_id,
        'ご予約ありがとうございます。\n\n' +
        '■ ' + fmt(start) + '〜（' + minutes + '分）\n' +
        '■ ' + menu + '\n' +
        '■ 担当：' + staff + '\n\n' +
        'ご予約の確認・変更はこちらから\nhttps://app.gait-pilates.com/\n\n' +
        'リラクゼーション＆ピラティス gait');
      sbRpc_('line_mark_sent', { p_secret: SHARED_SECRET, p_booking_id: Number(bookingId) || null,
        p_kind: 'confirm', p_profile: body.profile_id || null, p_body: '予約確定', p_ok: okLine });
    }

    MailApp.sendEmail(NOTIFY_EMAIL,
      '【gait】新規予約 ' + fmt(start) + ' ' + customer + ' さん',
      '新しい予約が入りました。\n\n' +
      '日時：' + fmt(start) + '〜（' + minutes + '分）\n' + desc +
      '\n\nGoogleカレンダーにも登録しました。');

    return ContentService.createTextOutput('ok');

  } catch (err) {
    MailApp.sendEmail(NOTIFY_EMAIL, '【gait】連携エラー', String(err) + '\n\n' + (e && e.postData ? e.postData.contents : ''));
    return ContentService.createTextOutput('error');
  }
}

/** 15分ごとに自動実行される本体 */
function syncCalendarToApp() {
  var now  = new Date();
  var from = new Date(now.getTime() - SYNC_BACK_DAYS    * 86400000);
  var to   = new Date(now.getTime() + SYNC_FORWARD_DAYS * 86400000);
  var res  = syncWindow(from, to);
  console.log('sync ' + fmtDay(from) + '〜' + fmtDay(to) + ' → ' + JSON.stringify(res));
  return res;
}

/** 指定した期間のカレンダーをアプリに取り込む */
function syncWindow(from, to) {
  var cal    = CalendarApp.getCalendarById(CALENDAR_ID);
  var events = cal.getEvents(from, to);
  var payload = [];

  for (var i = 0; i < events.length; i++) {
    var ev = events[i];
    if (ev.isAllDayEvent()) continue;              // 終日予定（誕生日・記念日など）は対象外
    var s = ev.getStartTime(), t = ev.getEndTime();
    payload.push({
      id: ev.getId() + '|' + s.getTime(),   // 繰り返し予定は全回で同じIDが返るため開始時刻を足して一意にする
      summary: ev.getTitle() || '',
      start: s.toISOString(),
      end:   t ? t.toISOString() : null,
      duration_min: t ? Math.round((t.getTime() - s.getTime()) / 60000) : null,
      description: (ev.getDescription() || '').slice(0, 500)
    });
  }

  var body = {
    p_secret: SHARED_SECRET,
    p_window_start: from.toISOString(),
    p_window_end:   to.toISOString(),
    p_events: payload
  };

  var r = UrlFetchApp.fetch(SB_URL + '/rest/v1/rpc/gcal_ingest', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'apikey': SB_KEY, 'Authorization': 'Bearer ' + SB_KEY },
    payload: JSON.stringify(body),
    muteHttpExceptions: true
  });

  var txt = r.getContentText();
  if (r.getResponseCode() >= 300) {
    throw new Error('gcal_ingest failed ' + r.getResponseCode() + ' ' + txt);
  }
  return { sent: payload.length, result: txt };
}

/** 過去分の一括取り込み（1回だけ手動実行）。2024年1月〜今日までを1ヶ月ずつ処理します */
function backfill2024() { backfillFrom(2024, 1); }

/** 2025年8月から取り込みたいときはこちら */
function backfill2025Aug() { backfillFrom(2025, 8); }

function backfillFrom(year, month) {
  var cur = new Date(year, month - 1, 1);
  var end = new Date();
  var log = [];
  while (cur < end) {
    var next = new Date(cur.getFullYear(), cur.getMonth() + 1, 1);
    try {
      var res = syncWindow(cur, next);
      log.push(fmtDay(cur) + ' → ' + res.result);
    } catch (err) {
      log.push(fmtDay(cur) + ' → ERROR ' + String(err));
    }
    Utilities.sleep(1000);
    cur = next;
  }
  console.log(log.join('\n'));
  MailApp.sendEmail(NOTIFY_EMAIL, '【gait】カレンダー一括取り込み 完了', log.join('\n'));
}

/** LINEへメッセージを1通送る。成功なら true */
function linePush_(to, text) {
  var token = lineToken_();
  if (!token) { console.log('LINE_TOKEN が未設定です'); return false; }
  var r = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({ to: to, messages: [{ type: 'text', text: String(text).slice(0, 4900) }] }),
    muteHttpExceptions: true
  });
  if (r.getResponseCode() >= 300) { console.log('LINE push 失敗 ' + r.getResponseCode() + ' ' + r.getContentText()); return false; }
  return true;
}

/** 受信したメッセージにその場で返信する（replyTokenは1回だけ・約1分で失効） */
function lineReply_(replyToken, text) {
  var token = lineToken_();
  if (!token) return false;
  var r = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/reply', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({ replyToken: replyToken, messages: [{ type: 'text', text: String(text).slice(0, 4900) }] }),
    muteHttpExceptions: true
  });
  return r.getResponseCode() < 300;
}

/** SupabaseのRPCを呼ぶ */
function sbRpc_(fn, params) {
  var r = UrlFetchApp.fetch(SB_URL + '/rest/v1/rpc/' + fn, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'apikey': SB_KEY, 'Authorization': 'Bearer ' + SB_KEY },
    payload: JSON.stringify(params),
    muteHttpExceptions: true
  });
  if (r.getResponseCode() >= 300) { console.log('RPC ' + fn + ' 失敗 ' + r.getResponseCode() + ' ' + r.getContentText()); return null; }
  try { return JSON.parse(r.getContentText()); } catch (err) { return null; }
}

/** LINEからのWebhookを処理する */
function handleLineWebhook_(body) {
  try {
    var evs = body.events || [];
    for (var i = 0; i < evs.length; i++) {
      var ev = evs[i];
      var uid = ev.source && ev.source.userId;
      if (!uid) continue;

      if (ev.type === 'follow') {
        lineReply_(ev.replyToken, LINE_WELCOME);
        continue;
      }

      if (ev.type === 'message' && ev.message && ev.message.type === 'text') {
        var txt = String(ev.message.text || '').trim();
        var code = txt.replace(/[\s　]/g, '');

        // 電話番号（10〜11桁の数字）なら、アプリ未登録の方でも連携できる
        if (/^0[0-9]{9,10}$/.test(code)) {
          var t = sbRpc_('line_link_by_tel', { p_secret: SHARED_SECRET, p_tel: code, p_line_user_id: uid });
          if (t && t.ok) {
            lineReply_(ev.replyToken,
              (t.name || 'お客様') + ' さん、連携が完了しました！\n\n' +
              '前日のお知らせをこちらにお送りします。\n' +
              '「予約」と送ると、今後のご予約と回数券の残りをいつでも確認できます。');
            MailApp.sendEmail(NOTIFY_EMAIL, '【gait】LINE連携（電話番号）',
              (t.name || '') + ' さんが電話番号でLINE連携しました。\n心当たりがない場合はオーナー画面で解除してください。');
          } else if (t && t.reason === 'duplicate') {
            lineReply_(ev.replyToken, '同じ電話番号のご登録が複数ありました。お手数ですが、スタジオでお声がけください。');
          } else if (t && t.reason === 'already_linked') {
            lineReply_(ev.replyToken, 'この電話番号はすでに別のLINEと連携済みです。スタジオでお声がけください。');
          } else {
            lineReply_(ev.replyToken,
              'ご登録が見つかりませんでした。\n' +
              'スタジオにお伝えいただいている電話番号を、ハイフンなしで送ってください。\n' +
              '例：09012345678');
          }
          continue;
        }

        // 8文字の英数字なら連携コードとして扱う
        if (/^[0-9A-Za-z]{8}$/.test(code)) {
          var res = sbRpc_('line_link', { p_secret: SHARED_SECRET, p_code: code, p_line_user_id: uid });
          if (res && res.ok) {
            lineReply_(ev.replyToken,
              (res.nickname || 'お客様') + ' さん、連携が完了しました！\n\n' +
              'これから、ご予約が入ったときの確認と、前日18時ごろのお知らせをこちらにお送りします。\n' +
              'いつでも解除できますので、不要になったら「配信停止」と送ってください。');
          } else {
            lineReply_(ev.replyToken,
              'コードが確認できませんでした。\n' +
              '有効期限は発行から30分です。アプリでもう一度コードを出して、お手数ですが送り直してください。\n' +
              'https://app.gait-pilates.com/');
          }
          continue;
        }

        // ▼ ここから下は「応答メッセージ」なので、何通送っても無料枠を消費しません
        if (/予約|よやく|確認|回数券|チケット|残り|次回/.test(txt)) {
          var st = sbRpc_('line_my_status', { p_secret: SHARED_SECRET, p_line_user_id: uid });
          if (!st || !st.ok) {
            lineReply_(ev.replyToken, 'まだ連携が済んでいないようです。\nアプリのマイページで連携コードを出して、こちらに送ってください。\nhttps://app.gait-pilates.com/');
          } else {
            var lines = [(st.name || 'お客様') + ' さん', ''];
            var bs = st.bookings || [];
            if (bs.length) {
              lines.push('■ 今後のご予約');
              for (var k = 0; k < bs.length; k++) lines.push('　' + fmt(new Date(bs[k].starts_at)) + '　' + bs[k].menu);
            } else {
              lines.push('■ 今後のご予約はありません');
            }
            var ts = st.tickets || [];
            if (ts.length) {
              lines.push('');
              lines.push('■ 回数券の残り');
              for (var q = 0; q < ts.length; q++) lines.push('　' + ts[q].left + ' / ' + ts[q].total + '回');

        }

        if (/アクセス|場所|地図|住所|行き方/.test(txt)) {
          lineReply_(ev.replyToken,
            'リラクゼーション＆ピラティス gait\n岡山市\n\n地図はこちら\nhttps://www.google.com/maps/search/?api=1&query=%E3%83%AA%E3%83%A9%E3%82%AF%E3%82%BC%E3%83%BC%E3%82%B7%E3%83%A7%E3%83%B3%EF%BC%86%E3%83%94%E3%83%A9%E3%83%86%E3%82%A3%E3%82%B9gait%20%E5%B2%A1%E5%B1%B1');
          continue;
        }

        if (/宿題|動画|エクササイズ|ホームワーク/.test(txt)) {
          lineReply_(ev.replyToken,
            '宿題動画はアプリのマイページからご覧いただけます。\nhttps://app.gait-pilates.com/\n\nログイン後「マイページ」→「宿題動画」です。');
          continue;
        }

        if (/配信再開|再開/.test(txt)) {
          sbRpc_('line_link', { p_secret: SHARED_SECRET, p_code: '__optin__', p_line_user_id: uid });
          lineReply_(ev.replyToken, 'お知らせの配信を再開しました。');
          continue;
        }

        if (/配信停止|停止|解除/.test(txt)) {
          sbRpc_('line_link', { p_secret: SHARED_SECRET, p_code: '__optout__', p_line_user_id: uid });

  } catch (err) {
    MailApp.sendEmail(NOTIFY_EMAIL, '【gait】LINE連携エラー', String(err));

}

/** 前日リマインド（毎日18時ごろに自動実行） */

  var list = sbRpc_('line_due_reminders', { p_secret: SHARED_SECRET }) || [];
  var sent = 0;
  for (var i = 0; i < list.length; i++) {
    var b = list[i];
    var d = new Date(b.starts_at);
    var msg = (b.name || 'お客様') + ' さん\n\n' +
      'あすのご予約のお知らせです。\n\n' +

      'ご都合が変わった場合は、お早めにご連絡ください。\n' +
      'https://app.gait-pilates.com/\n\n' +
      'リラクゼーション＆ピラティス gait';
    var ok = linePush_(b.line_user_id, msg);
    sbRpc_('line_mark_sent', { p_secret: SHARED_SECRET, p_booking_id: b.booking_id,
      p_kind: 'reminder', p_profile: b.profile_id, p_body: msg, p_ok: ok });
    if (ok) sent++;
    Utilities.sleep(200);
  }
  var u = sbRpc_('line_month_usage', { p_secret: SHARED_SECRET }) || { used: 0, cap: 200 };
  console.log('前日リマインド ' + sent + '/' + list.length + '件 送信　今月の無料枠 ' + u.used + '/' + u.cap + '通');
  if (u.used >= 170) {
    MailApp.sendEmail(NOTIFY_EMAIL, '【gait】LINEの無料枠が残りわずかです',
      '今月の送信数：' + u.used + ' / ' + u.cap + '通\n\n' +
      '190通に達した時点で自動送信は止まります（超過エラーを避けるため）。\n' +
      'お客様からの「予約」などへの返信は無料なので、そちらは止まりません。');
  }
  return { total: list.length, sent: sent, usage: u };
}

/** 一度だけ実行してください。15分ごとの自動取り込みが始まります */
function setupTriggers() {
  var ts = ScriptApp.getProjectTriggers();
  for (var i = 0; i < ts.length; i++) {
    if (ts[i].getHandlerFunction() === 'syncCalendarToApp') ScriptApp.deleteTrigger(ts[i]);
  }
  for (var j = 0; j < ts.length; j++) {
    if (ts[j].getHandlerFunction() === 'sendTomorrowReminders') ScriptApp.deleteTrigger(ts[j]);
  }
  ScriptApp.newTrigger('syncCalendarToApp').timeBased().everyMinutes(15).create();
  ScriptApp.newTrigger('sendTomorrowReminders').timeBased().atHour(18).everyDays(1).inTimezone('Asia/Tokyo').create();
  console.log('15分ごとの自動同期と、毎日18時の前日リマインドを設定しました');
}

/** 自動同期を止めたいとき */
function removeTriggers() {
  var ts = ScriptApp.getProjectTriggers();
  for (var i = 0; i < ts.length; i++) ScriptApp.deleteTrigger(ts[i]);
  console.log('すべてのトリガーを削除しました');
}
